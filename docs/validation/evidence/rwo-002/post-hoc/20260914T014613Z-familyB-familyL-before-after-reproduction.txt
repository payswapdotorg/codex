#!/usr/bin/env bash

#
# Surface: FlowMart fixture (port 4105) — HTML form twin (urlencoded, the
# browser-path equivalent: the /ui/installs/configure form POSTs the textarea
# as a STRING) + JSON API twins (/api/installs/*). Demo personas only; tokens
# are kept in shell variables and never printed.
set -uo pipefail
BASE_URL=http://localhost:4105
JAR=$(mktemp)

say()  { echo; echo "=================================================================="; echo "$1"; echo "=================================================================="; }

# login iris.chen (Northwind org_admin) — API twin (token kept out of output)
T=$(curl -s "$BASE_URL/api/login" -H 'content-type: application/json' \
  -d '{"username":"iris.chen","password":"demo-pass-iris"}' \
  | sed -n 's/.*"token": "\([^"]*\)".*/\1/p')
[ -n "$T" ] && echo "# login iris.chen -> ok (token captured, not printed)" || echo "# login FAILED"

# login iris.chen via the /login HTML form -> session cookie (browser semantics)
curl -s -o /dev/null -c "$JAR" -X POST "$BASE_URL/login" \
  --data-urlencode 'username=iris.chen' --data-urlencode 'password=demo-pass-iris'
[ -s "$JAR" ] && echo "# login form -> session cookie set (jar captured)" || echo "# login form FAILED"

# show one install record (config + history) via the API twin
show_install() {
  curl -s "$BASE_URL/api/installs?token=$T" \
    | node -e 'let d="";process.stdin.on("data",c=>d+=c).on("end",()=>{const j=JSON.parse(d);const i=(j.installs||[]).find(x=>x.id==="ins-0402")||j.installs[j.installs.length-1];if(!i){console.log("  (no installs)");return;}console.log("  install "+i.id+" config: "+JSON.stringify(i.config));const h=i.history[i.history.length-1];console.log("  last history entry: "+JSON.stringify(h));})'
}

say "SECTION 1 — seed to the sweep-golden state (reset; iris installs pk-103 with object config {\"zones\":\"A\"} -> ins-0402)"
curl -s -X POST "$BASE_URL/api/reset" -o /dev/null
curl -s -w '\nHTTP %{http_code}\n' "$BASE_URL/api/installs/install?token=$T" -H 'content-type: application/json' \
  -d '{"packageId":"pk-103","expectedVersion":"1.2.0","config":{"zones":"A"}}' | head -8
show_install

say "SECTION 2 — FAMILY B: browser-path equivalent (form twin, urlencoded STRING config) — POST /ui/installs/configure config={\"opsChannel\":\"east\"}"
echo '$ curl -b <session-cookie> -X POST /ui/installs/configure (Content-Type: application/x-www-form-urlencoded; installId=ins-0402, config={"opsChannel":"east"})'
curl -s -o /dev/null -b "$JAR" -X POST "$BASE_URL/ui/installs/configure" \
  -H 'Content-Type: application/x-www-form-urlencoded' \
  --data-urlencode 'installId=ins-0402' --data-urlencode 'config={"opsChannel":"east"}' \
  -w 'HTTP %{http_code} -> redirect: %{redirect_url}\n'
echo '# /installs page as the browser would render it (flash + Config cell + history rows):'
curl -s -b "$JAR" "$BASE_URL/installs" | grep -oE '<div class="flash[^>]*>[^<]*</div>|<dt>Config</dt><dd>[^<]*<code>[^<]*</code>|configured [^<]*' | head -6
echo '# state check via API twin:'
show_install

say "SECTION 3 — API twins on /api/installs/configure (JSON bodies)"
echo '== 3a. string config (API twin of the form semantics): config "{\"apiTwin\":\"string-config\"}"'
curl -s -w '\nHTTP %{http_code}\n' "$BASE_URL/api/installs/configure?token=$T" -H 'content-type: application/json' \
  -d '{"installId":"ins-0402","config":"{\"apiTwin\":\"string-config\"}"}' | head -10
echo '== 3b. INVALID JSON string: config "{oops"'
curl -s -w '\nHTTP %{http_code}\n' "$BASE_URL/api/installs/configure?token=$T" -H 'content-type: application/json' \
  -d '{"installId":"ins-0402","config":"{oops"}' | head -10
echo '== 3c. RESERVED identity-impersonating keys: config {"version":"9.9.9","digest":"sha256:fake","targetVersion":"9.9.9","manifest":"attacker-manifest"}'
curl -s -w '\nHTTP %{http_code}\n' "$BASE_URL/api/installs/configure?token=$T" -H 'content-type: application/json' \
  -d '{"installId":"ins-0402","config":{"version":"9.9.9","digest":"sha256:fake","targetVersion":"9.9.9","manifest":"attacker-manifest"}}' | head -10
echo '== 3d. object twin (regression check): config {"opsChannel":"object-path"}'
curl -s -w '\nHTTP %{http_code}\n' "$BASE_URL/api/installs/configure?token=$T" -H 'content-type: application/json' \
